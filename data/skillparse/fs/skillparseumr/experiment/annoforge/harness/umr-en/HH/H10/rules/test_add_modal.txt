a = '# sentence level graph:\n(s1s / say-01\n    :ARG0 (s1p / person\n        :ARG0-of (s1r / research-01))\n    :ARG1 (s1d / die-01\n        :ARG1 (s1m / man)\n        :quote s1s\n        :aspect performance)\n    :aspect performance)\n\n# alignment:\ns1s: 1-1\n\n# document level annotation:\n(s1s0 / sentence\n    :modal ((root :modal author)\n            (author :full-affirmative s1s)))'
out = run(a, {'index': 1, 'text': ''})
assert '(s1p :full-affirmative s1d)' in out
assert '(author :full-affirmative s1p)' in out
assert 'full-affirmative s1r' not in out
assert out.count('(root :modal author)') == 1
