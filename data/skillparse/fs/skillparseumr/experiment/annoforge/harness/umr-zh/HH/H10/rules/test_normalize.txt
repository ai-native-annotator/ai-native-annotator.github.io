ANN = '# sentence level graph:\n(s1x / plan-01\n    :ARG0 (s1p / person\n        :ref-person 3rd\n        :ref-number Singular)\n    :aspect State\n    :modstr FullAff)\n\n# alignment:\ns1x: 1-1\ns1p: 0-0\n\n# document level annotation:\n(s1s0 / sentence\n    :temporal ((PAST_REF :overlap s1x))\n    :modal ((AUTH :FullAff s1x)))'
out = run(ANN, {'index': 1, 'text': ''})
assert ':modstr' not in out
assert 'PAST_REF' not in out
assert ':temporal' not in out
assert '(author :full-affirmative s1x)' in out
assert ':refer-person 3rd' in out and ':refer-number singular' in out
assert ':aspect state' in out
ANN2 = '# sentence level graph:\n(s1x / change-01\n    :aspect performance\n    :modstr NeutNeg\n    :polarity -)\n\n# alignment:\ns1x: 1-1\n\n# document level annotation:\n(s1s0 / sentence)'
out2 = run(ANN2, {'index': 1, 'text': ''})
assert '(author :neutral-negative s1x)' in out2
assert ':modal' in out2
assert out2.count('(') == out2.count(')')
ANN3 = '# sentence level graph:\n(s1x / a\n    :aspect process\n    :ARG1 (s1y / b\n        :aspect process))\n\n# alignment:\ns1x: 1-1\ns1y: 2-2\n\n# document level annotation:\n(s1s0 / sentence\n    :temporal ((s1x :overlap s1y)\n               (DCT :before s1y)\n               (DCT :before s1x))\n    :modal ((AUTH :Unsp s1y)))'
out3 = run(ANN3, {'index': 1, 'text': ''})
assert '(s1x :overlap s1y)' in out3
assert 'before s1y' not in out3
assert '(document-creation-time :before s1x)' in out3
assert '(author :unspecified s1y)' in out3
assert '(author :full-affirmative s1x)' in out3
ANN4 = 'garbage without sections'
assert run(ANN4, {'index': 1, 'text': ''}) == ANN4
import inspect
assert run.__doc__ and run.__doc__.strip()
sig = inspect.signature(run)
assert all(p.annotation is not inspect.Parameter.empty for p in sig.parameters.values())
assert sig.return_annotation is not inspect.Signature.empty
