a = '# sentence level graph:\n(s1x / a)\n\n# alignment:\ns1x: 1-1\n\n# document level annotation:\n(s1s0 / sentence\n    :modal ((author :full-affirmative s1x)))\n'
r = run(a, {'index': 1, 'text': 'x'})
assert '((root :modal author)' in r and '(author :full-affirmative s1x)' in r
b = '# sentence level graph:\n(s1x / a)\n\n# alignment:\ns1x: 1-1\n\n# document level annotation:\n(s1s0 / sentence\n    :temporal ((document-creation-time :before s1x)))\n'
r2 = run(b, {'index': 1, 'text': 'x'})
assert ':modal ((root :modal author))' in r2 and r2.count('(s1s0') == 1
c = '# sentence level graph:\n(s1x / a)\n\n# alignment:\ns1x: 1-1\n\n# document level annotation:\n(s1s0 / sentence\n    :modal ((root :modal author)\n            (author :full-affirmative s1x)))\n'
assert run(c, {'index': 1, 'text': 'x'}) == c
